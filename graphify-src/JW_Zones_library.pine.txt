// JW_Zones — ط4-أ: مناطق الدعم/المقاومة (الدرس L2-02) — خوارزميّة التزويج الموحّدة النقيّة
//   مُستخرَجة من J-Waves M2 (فرع m3/gaps-and-scale).
//
//   القاعدة (حسم المستخدم 2026-07-14 — يَنسخ «نفس النوع فقط» المستخرَج سابقًا من الفيديو):
//     «مستطيل بين كلّ خطّين متقاربين سعريًّا» — **بأيّ نوع** (قمّة/قاع)، الأقرب سعريًّا (الأحدث أوّلًا)،
//     **الجوار** (لا مستوى بينهما سعريًّا ⇒ متجاوران)، **لا تداخل** (كامل+جزئيّ)، **لا حدّ أدنى** للعرض.
//     🔴 الاستثناء الوحيد: **قمّة + تصحيحها المباشر** = طرفان معاكسا النوع من **نفس الدرجة** متتاليان
//        زمنيًّا (لا طرف نفس-الدرجة بينهما زمنيًّا = موجة واحدة) ⇒ لا منطقة بينهما.
//     البركة موحّدة (تُبنى في المؤشّر): المتوسطة الفعّالة + تاريخ التاريخ (المُغطّاة) + الكبيرة ⇒ لا استعارة منفصلة.
//
//   الحدّ: خوارزميّة نقيّة تعيد حوافّ المناطق [zTop, zBot, zLeft, usedZ]. الرسم/الخطوط/الألوان تبقى في المؤشّر.
//   eLvl=مستوى الجسم · eTp=قمّة؟ · eBar=بار الطرف · eDeg=الدرجة (0 متوسطة · 1 كبيرة). OHLC/bar_index تعمل داخلها (v6).
// ════════════════════════════════════════════════════════════════════════════
library("JW_Zones", overlay = true)

// ─── التزويج الموحّد — بين كلّ طرفين متجاورين سعريًّا (أيّ نوع) منطقة، إلّا قمّة+تصحيحها المباشر ───
export f_pairZones(array<float> eLvl, array<bool> eTp, array<int> eBar, array<int> eDeg) =>
    int en = array.size(eLvl)
    array<bool>  usedZ = array.new<bool>(en, false)
    array<float> zTop  = array.new<float>()
    array<float> zBot  = array.new<float>()
    array<int>   zLeft = array.new<int>()
    if en >= 1
        for oi = 0 to en - 1
            int aa = en - 1 - oi                                    // الأحدث أوّلًا
            if not array.get(usedZ, aa)
                int   bb = -1
                float bd = 1e18
                for c = 0 to en - 1
                    if c != aa and not array.get(usedZ, c)
                        float d   = math.abs(array.get(eLvl, aa) - array.get(eLvl, c))
                        float loB = math.min(array.get(eLvl, aa), array.get(eLvl, c))
                        float hiB = math.max(array.get(eLvl, aa), array.get(eLvl, c))
                        bool between = false                        // الجوار السعريّ: لا مستوى بينهما ⇒ متجاوران
                        for k = 0 to en - 1
                            if k != aa and k != c and array.get(eLvl, k) > loB and array.get(eLvl, k) < hiB
                                between := true
                        bool directCorr = false                    // 🔴 قمّة + تصحيحها المباشر: معاكسا النوع + نفس الدرجة + متتاليان زمنيًّا
                        if array.get(eTp, c) != array.get(eTp, aa) and array.get(eDeg, c) == array.get(eDeg, aa)
                            int loT = math.min(array.get(eBar, aa), array.get(eBar, c))
                            int hiT = math.max(array.get(eBar, aa), array.get(eBar, c))
                            bool timeBetween = false
                            for k = 0 to en - 1
                                if k != aa and k != c and array.get(eDeg, k) == array.get(eDeg, aa) and array.get(eBar, k) > loT and array.get(eBar, k) < hiT
                                    timeBetween := true
                            directCorr := not timeBetween          // لا طرف نفس-الدرجة بينهما زمنيًّا ⇒ موجة واحدة (قمّة↔تصحيحها)
                        if not between and not directCorr and d < bd
                            bd := d
                            bb := c
                if bb >= 0
                    float cT = math.max(array.get(eLvl, aa), array.get(eLvl, bb))
                    float cB = math.min(array.get(eLvl, aa), array.get(eLvl, bb))
                    bool contain = false                            // 🔴 لا منطقة داخل نطاق منطقة أخرى (تقاطع؛ يسمح بتلامس الحوافّ)
                    int zc = array.size(zTop)
                    if zc > 0
                        for k = 0 to zc - 1
                            if array.get(zBot, k) < cT and cB < array.get(zTop, k)
                                contain := true
                    if not contain
                        array.set(usedZ, aa, true)
                        array.set(usedZ, bb, true)
                        int leftBar = math.min(array.get(eBar, aa), array.get(eBar, bb))
                        array.push(zTop, cT)
                        array.push(zBot, cB)
                        array.push(zLeft, leftBar)
    [zTop, zBot, zLeft, usedZ]
